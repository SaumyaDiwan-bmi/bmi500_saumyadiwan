import random

# create a function to compute the dot product of two vectors using a for loop
def dot_product(vec1, vec2):
    """
    Compute the dot product of two vectors using a for loop.

    Args:
        vec1 (list): First vector (list of numbers)
        vec2 (list): Second vector (list of numbers)

    Returns:
        float/int: The dot product of vec1 and vec2
    """
    # guard: both inputs must actually be lists/tuples, not None or something else
    if not isinstance(vec1, (list, tuple)) or not isinstance(vec2, (list, tuple)):
        raise TypeError(
            f"Both vectors must be lists or tuples, got {type(vec1).__name__} and {type(vec2).__name__}"
        )

    # guard: reject empty vectors explicitly rather than silently returning 0
    if len(vec1) == 0 or len(vec2) == 0:
        raise ValueError("Vectors must not be empty")

    # guard: vectors must be the same length to compute a dot product
    if len(vec1) != len(vec2):
        raise ValueError(
            f"Vectors must be the same length, got {len(vec1)} and {len(vec2)}"
        )

    # accumulator for the running sum of element-wise products
    result = 0

    # loop through each index, multiply corresponding elements, and add to result
    for i in range(len(vec1)):
        # guard: catch non-numeric elements with a clear, indexed error message
        try:
            result += vec1[i] * vec2[i]
        except TypeError:
            raise TypeError(
                f"Non-numeric element at index {i}: {vec1[i]!r} * {vec2[i]!r}"
            )

    return result
