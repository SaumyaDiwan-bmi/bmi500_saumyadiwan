def dot_product(vec1, vec2):
    """
    Compute the dot product of two vectors using a for loop.
    
    Args:
        vec1 (list): First vector (list of numbers)
        vec2 (list): Second vector (list of numbers)
    
    Returns:
        float/int: The dot product of vec1 and vec2
    """
    #guard: ensuring both inputs be lists or tuples
    if not isinstance(vec1, (list, tuple)) or not isinstance(vec2, (list, tuple)):
        raise TypeError(
            f"Input vectors must be lists or tuples, got {type(vec1).__name__} and {type(vec2).__name__}"
        )
    #guard: checking if vector is empty
    if len(vec1) == 0 or len(vec2) == 0:
        raise ValueError("Vectors must not be empty")
    
    #guard: vectors must be the same length to compute a dot product
    if len(vec1) != len(vec2):
        raise ValueError("Vectors must be the same length")

    # accumulator for the running sum of element-wise products
    result = 0

    # loop through each index, multiply corresponding elements, and add to result
    for i in range(len(vec1)):
        result += vec1[i] * vec2[i]

    return result

import random


def dot_product(vec1, vec2):
    """
    Compute the dot product of two vectors using a for loop.

    Args:
        vec1 (list or tuple): First vector.
        vec2 (list or tuple): Second vector.

    Returns:
        int or float: Dot product of the two vectors.
    """

    if not isinstance(vec1, (list, tuple)) or not isinstance(vec2, (list, tuple)):
        raise TypeError("Input vectors must be lists or tuples")

    if len(vec1) == 0 or len(vec2) == 0:
        raise ValueError("Vectors must not be empty")

    if len(vec1) != len(vec2):
        raise ValueError("Vectors must be the same length")

    result = 0

    for x, y in zip(vec1, vec2):
        result += x * y

    return result


def matvec_product(matrix, vec):
    """
    Compute the product of a matrix and vector using dot_product().

    Args:
        matrix (list of lists): Matrix represented by rows.
        vec (list or tuple): Vector to multiply by the matrix.

    Returns:
        list: Resulting matrix-vector product.
    """

    if not isinstance(matrix, (list, tuple)):
        raise TypeError("Matrix must be a list or tuple")

    if not isinstance(vec, (list, tuple)):
        raise TypeError("Vector must be a list or tuple")

    if len(matrix) == 0:
        raise ValueError("Matrix must not be empty")

    if not all(isinstance(row, (list, tuple)) for row in matrix):
        raise TypeError("Every matrix row must be a list or tuple")

    # Ensure the matrix is rectangular.
    row_lengths = {len(row) for row in matrix}

    if len(row_lengths) != 1:
        raise ValueError("All matrix rows must have the same length")

    if len(matrix[0]) == 0:
        raise ValueError("Matrix rows must not be empty")

    if len(matrix[0]) != len(vec):
        raise ValueError(
            "Number of matrix columns must equal vector length"
        )

    result = []

    for row in matrix:
        result.append(dot_product(row, vec))

    return result


def main():
    size = 1000

    # Make the random test reproducible.
    random.seed(25)

    # First verify correctness using a simple known example.
    assert matvec_product(
        [[1, 2], [3, 4]],
        [5, 6]
    ) == [17, 39]

    matrix = [
        [random.random() for _ in range(size)]
        for _ in range(size)
    ]

    vec = [random.random() for _ in range(size)]

    try:
        result = matvec_product(matrix, vec)

    except (TypeError, ValueError) as e:
        print(f"Matrix-vector product failed: {e}")
        raise SystemExit(1)

    print("First 5 elements of the result:", result[:5])
    print("Length of result vector:", len(result))
